#Extract final placement positions from grid version
extract_final_placements <- function(row, x_field, y_field,
                                     source_col = "move_log") {

  raw <- row[[source_col]]
  if (is.na(raw) || raw == "[]") return(NULL)

  parsed <- tryCatch(fromJSON(raw), error = function(e) NULL)
  if (is.null(parsed) || !is.data.frame(parsed)) return(NULL)

  if (!all(c(x_field, y_field) %in% names(parsed))) return(NULL)

  img_col <- if ("image_name" %in% names(parsed)) "image_name"
             else if ("image" %in% names(parsed)) "image"
             else return(NULL)

  parsed$posX <- suppressWarnings(as.numeric(parsed[[x_field]]))
  parsed$posY <- suppressWarnings(as.numeric(parsed[[y_field]]))

  # Only relevant for move_log (history) — no-op for placements (already final)
  if ("timestamp" %in% names(parsed)) {
    parsed <- parsed %>%
      mutate(timestamp = suppressWarnings(as.numeric(timestamp))) %>%
      group_by(.data[[img_col]]) %>%
      arrange(timestamp) %>%
      slice_tail(n = 1) %>%
      ungroup()
  }

  parsed %>%
    transmute(
      participant_id = row$participant_id,
      block          = suppressWarnings(as.numeric(row$block)),
      trial          = suppressWarnings(as.numeric(row$trial)),
      image_name     = .data[[img_col]],
      posX, posY
    )
}
# Tag category / base_name / set from image filename.
# Image filenames look like "happy_1.png", "horse_2.jpg", etc.
tag_categories <- function(df) {
  df %>%
    mutate(
      image_stem = str_remove(image_name, "\\.[A-Za-z0-9]+$"),
      base_name  = str_remove(image_stem,  "_\\d+$"),
      set        = str_extract(image_stem, "\\d+$") %>% as.integer(),
      category   = case_when(
        str_detect(base_name,
                   "happy|sad|angry|fearful|surprise|disgust") ~ "emotions",
        str_detect(base_name,
                   "horse|lion|wolf|fish|squirrel|sheep|peacock|pigeon|fox|butterfly|ladybug|dolphin") ~ "animals",
        str_detect(base_name,
                   "tree|flower|grass|leaf|apple|broccoli|blueberry|acorn|cherry|cactus|cabbage|peanut") ~ "plants"
      )
    )
}
# Normalize "p###" participant IDs from raw column.
normalize_pid <- function(x) {
  str_extract(as.character(x), "\\d+") %>%
    str_pad(width = 3, pad = "0") %>%
    str_c("p", .)
}

# Load all CSVs from a folder, normalize IDs.
load_folder <- function(folder) {
  list.files(folder, pattern = "\\.csv$", full.names = TRUE) %>%
    map_dfr(~ read_csv(.x, col_types = cols(.default = "c"))) %>%
    mutate(participant_id = normalize_pid(participant_id))
}

# Build final_placements.
build_placements <- function(folder, trial_type_label, x_field, y_field,
                             source_col = "move_log") {
  all_data <- load_folder(folder)

  trials <- all_data %>%
    filter(trial_type == trial_type_label,
           !is.na(block), block != "null")

  if (nrow(trials) == 0) {
    stop("No rows matched trial_type == '", trial_type_label,
         "'. trial_type values present: ",
         paste(unique(all_data$trial_type), collapse = ", "))
  }

  out <- trials %>%
    split(seq_len(nrow(.))) %>%
    map_dfr(~ extract_final_placements(.x, x_field, y_field, source_col))

  if (nrow(out) == 0) {
    stop("Extraction returned 0 placements from '", source_col, "'. ",
         "Check x_field/y_field ('", x_field, "', '", y_field,
         "') against the JSON keys in that column.")
  }

  out %>%
    mutate(trial = suppressWarnings(as.numeric(trial))) %>%
    filter(!is.na(trial), trial > 0) %>%
    tag_categories()
}
# Pairwise-distance vector from a set of coords, sorted by base_name
compute_dist_vec <- function(df) {
  coords <- df %>%
    arrange(base_name) %>%
    select(posX, posY) %>%
    as.matrix()
  as.vector(dist(coords))
}
# Test-retest consistency per (participant, category).
#
# For each trial, average duplicate-emotion picture positions to one
# point per concept (no-op for animal/plant since they have 1 pic/concept).
# Then correlate trial 1 vs trial 2 distance vectors.
#
# Returns one row per (participant_id, category) with column `consistency`.
compute_consistency <- function(final_placements) {

  trial_positions <- final_placements %>%
    group_by(participant_id, category, trial, base_name) %>%
    summarise(posX = mean(posX), posY = mean(posY), .groups = "drop")

  trial_dist <- trial_positions %>%
    group_by(participant_id, category, trial) %>%
    nest() %>%
    mutate(dist_vec = map(data, compute_dist_vec)) %>%
    select(-data)

  trial_dist %>%
    group_by(participant_id, category) %>%
    summarise(
      consistency = if (length(dist_vec) < 2) NA_real_
                    else cor(dist_vec[[1]], dist_vec[[2]],
                             use = "complete.obs"),
      n_trials    = length(dist_vec),
      .groups     = "drop"
    )
}
# Emotions-only: within-trial duplicate clustering.
# Within each emotion trial, correlate set-1 vs set-2 distance vectors,
# then average across the participant's emotion trials.
compute_within_trial <- function(final_placements) {
  final_placements %>%
    filter(category == "emotions") %>%
    group_by(participant_id, trial, set) %>%
    nest() %>%
    mutate(dist_vec = map(data, compute_dist_vec)) %>%
    select(-data) %>%
    group_by(participant_id, trial) %>%
    summarise(
      within_cor = if (length(dist_vec) < 2) NA_real_
                   else cor(dist_vec[[1]], dist_vec[[2]],
                            use = "complete.obs"),
      .groups = "drop"
    ) %>%
    group_by(participant_id) %>%
    summarise(
      within_trial_consistency = mean(within_cor, na.rm = TRUE),
      n_trials = sum(!is.na(within_cor)),
      .groups = "drop"
    )
}
# Cluster the concepts in one trial, return cluster labels in base_name order.
cluster_trial <- function(df, k) {
  coords <- df %>% arrange(base_name) %>% select(posX, posY) %>% as.matrix()
  hc <- hclust(dist(coords), method = "ward.D2")
  cutree(hc, k = k)
}

# Cluster-based consistency: ARI between trial 1 and trial 2 cluster labels.
compute_cluster_consistency <- function(final_placements, k = 3) {

  trial_positions <- final_placements %>%
    group_by(participant_id, category, trial, base_name) %>%
    summarise(posX = mean(posX), posY = mean(posY), .groups = "drop")

  trial_clusters <- trial_positions %>%
    group_by(participant_id, category, trial) %>%
    nest() %>%
    mutate(cluster = map(data, ~ cluster_trial(.x, k = k))) %>%
    select(-data)

  trial_clusters %>%
    group_by(participant_id, category) %>%
    summarise(
      cluster_consistency = if (length(cluster) < 2) NA_real_
                            else mclust::adjustedRandIndex(cluster[[1]],
                                                          cluster[[2]]),
      n_trials = length(cluster),
      .groups = "drop"
    )
}

compute_cat_org <- function(final_placements, subcategories) {

  positions <- final_placements %>%
    group_by(participant_id, category, trial, base_name) %>%
    summarise(posX = mean(posX), posY = mean(posY), .groups = "drop") %>%
    left_join(subcategories, by = "base_name")

  ratio_one_trial <- function(df) {
    if (n_distinct(df$subcategory) < 2) return(NA_real_)
    d <- as.matrix(dist(as.matrix(df %>% select(posX, posY))))
    same <- outer(df$subcategory, df$subcategory, "==")
    diag(same) <- NA
    within  <- mean(d[same == TRUE],  na.rm = TRUE)
    between <- mean(d[same == FALSE], na.rm = TRUE)
    between / within
  }

  positions %>%
    group_by(participant_id, category, trial) %>%
    nest() %>%
    mutate(cat_org = map_dbl(data, ratio_one_trial)) %>%
    select(-data) %>%
    group_by(participant_id, category) %>%
    summarise(cat_org = mean(cat_org, na.rm = TRUE), .groups = "drop")
}
